# ============================================================
# 2. NUTRITIONAL SCORE FUNCTION
# ============================================================

def compute_score(item, alpha=1.0, beta=2.0, gamma=0.5):

    score = (
        alpha * item["cal"] +
        beta * item["prot"] +
        alpha * item["carb"] +
        gamma * item["fat"] +
        alpha * item["calcium"] +
        alpha * item["zinc"] +
        alpha * item["folate"]
    )

    return score


# ============================================================
# 3. DYNAMIC PROGRAMMING OPTIMIZATION
# ============================================================

def optimize_meal(categories, max_capacity):

    dp = {
        0: (0, [])
    }

    for group in categories:

        new_dp = {}

        for current_weight, (current_score, selected_items) in dp.items():

            for item in group:

                new_weight = current_weight + item["weight"]

                if new_weight <= max_capacity:

                    new_score = current_score + compute_score(item)

                    if (
                        new_weight not in new_dp
                        or new_score > new_dp[new_weight][0]
                    ):

                        new_dp[new_weight] = (
                            new_score,
                            selected_items + [item]
                        )

        dp = new_dp

    if not dp:
        return None, None

    best_weight, (best_score, best_items) = max(
        dp.items(),
        key=lambda x: x[1][0]
    )

    return best_score, best_items



# ============================================================
# 4. RUN OPTIMIZATION (MAXIMUM CAPACITY: 500 g)
# ============================================================

score, selected_items = optimize_meal(food_groups, max_capacity)
